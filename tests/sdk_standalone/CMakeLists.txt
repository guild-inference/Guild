cmake_minimum_required(VERSION 3.24)
project(guild_sdk_contract LANGUAGES C CXX)
find_package(GuildKernelSDK 1.4 REQUIRED CONFIG)
add_library(sdk_c_plugin SHARED plugin.c)
guild_configure_gkernel_plugin(sdk_c_plugin)
set_target_properties(sdk_c_plugin PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
add_executable(sdk_contract host.cpp)
target_compile_features(sdk_contract PRIVATE cxx_std_17)
target_link_libraries(sdk_contract PRIVATE Guild::KernelSDK ${CMAKE_DL_LIBS})
enable_testing()
add_test(NAME installed_c_cpp_abi COMMAND sdk_contract $<TARGET_FILE:sdk_c_plugin>)
